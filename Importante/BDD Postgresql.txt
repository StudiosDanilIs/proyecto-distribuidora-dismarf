-- 1. Tabla de Roles (Para separar Administradores de Supervisores de Almacén)
CREATE TABLE roles (
    id_rol SERIAL PRIMARY KEY,
    nombre VARCHAR(50) NOT NULL UNIQUE,
    descripcion TEXT
);

-- 2. Tabla de Usuarios (El personal de Dismarf que usará la App Móvil)
CREATE TABLE usuarios (
    id_usuario SERIAL PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    id_rol INT REFERENCES roles(id_rol),
    creado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 3. Tabla de Cavas Cuarto (Los lugares físicos que vamos a monitorear con los ESP32)
CREATE TABLE cavas (
    id_cava SERIAL PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    ubicacion VARCHAR(150),
    temp_minima_permitida DECIMAL(5,2) NOT NULL,
    temp_maxima_permitida DECIMAL(5,2) NOT NULL,
    estado VARCHAR(20) DEFAULT 'Activa'
);

-- 4. Tabla de Inventario (Los productos perecederos dentro de cada cava)
CREATE TABLE inventario (
    id_lote SERIAL PRIMARY KEY,
    producto VARCHAR(150) NOT NULL,
    cantidad INT NOT NULL,
    fecha_ingreso DATE NOT NULL,
    fecha_vencimiento DATE NOT NULL,
    id_cava INT REFERENCES cavas(id_cava),
    estado_calidad VARCHAR(50) DEFAULT 'Óptimo'
);

CREATE TABLE alertas (
    id SERIAL PRIMARY KEY,
    cava_id INT REFERENCES cavas(id) ON DELETE CASCADE,
    tipo VARCHAR(50), -- Ej: 'TEMPERATURA_ALTA', 'PUERTA_ABIERTA', 'DESCONEXION'
    mensaje TEXT,
    valor_registrado NUMERIC(5,2),
    fecha TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    resuelta BOOLEAN DEFAULT FALSE
);

CREATE TABLE auditorias (
    id SERIAL PRIMARY KEY,
    usuario_id INT REFERENCES usuarios(id),
    cava_id INT REFERENCES cavas(id),
    fecha_inspeccion TIMESTAMP,
    estado_limpieza VARCHAR(50), -- Ej: 'Excelente', 'Regular', 'Deficiente'
    observaciones TEXT,
    sincronizado_el TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

ALTER TABLE cavas 
ADD COLUMN tipo_producto VARCHAR(100) DEFAULT 'Mixto',
ADD COLUMN capacidad_ocupada INT DEFAULT 0,
ADD COLUMN ultimo_mantenimiento DATE DEFAULT CURRENT_DATE;

CREATE TABLE movimientos_inventario (
    id SERIAL PRIMARY KEY,
    cava_id INTEGER REFERENCES cavas(id),
    producto VARCHAR(255),
    tipo_movimiento VARCHAR(50),
    cantidad NUMERIC,
    unidad VARCHAR(50),
    fecha TIMESTAMP
);

ALTER TABLE usuarios 
ADD COLUMN activo BOOLEAN DEFAULT TRUE;

CREATE TABLE bitacora (
    id_bitacora SERIAL PRIMARY KEY,
    id_usuario INT REFERENCES usuarios(id_usuario) ON DELETE SET NULL,
    nombre_usuario VARCHAR(100),
    accion VARCHAR(50), -- Ej: 'CREAR', 'ACTUALIZAR', 'ELIMINAR', 'LOGIN'
    modulo VARCHAR(50), -- Ej: 'CAVAS', 'INVENTARIO', 'USUARIOS'
    detalle TEXT,       -- Ej: "Registró 50 Kg de Pollo"
    ip_origen VARCHAR(45),
    fecha TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
